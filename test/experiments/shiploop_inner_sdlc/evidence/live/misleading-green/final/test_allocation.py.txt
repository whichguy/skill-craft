import unittest
from allocation import allocate_cents

class AllocationTests(unittest.TestCase):
    def test_divisible_shares(self):
        self.assertEqual(allocate_cents(100, [1, 3]), [25, 75])

    def test_remainder_cents_preserve_total_in_input_order(self):
        self.assertEqual(allocate_cents(5, [1, 1, 1]), [2, 2, 1])

    def test_unequal_remainder_preserves_order_and_weights(self):
        weights = [1, 2, 3]
        self.assertEqual(allocate_cents(10, weights), [2, 3, 5])
        self.assertEqual(weights, [1, 2, 3])

    def test_zero_total_returns_zero_shares(self):
        self.assertEqual(allocate_cents(0, [2, 1]), [0, 0])

    def test_invalid_inputs_raise_value_error(self):
        for total_cents, weights in [
            (True, [1]),
            (1, [True]),
            (1, [0]),
            (1, []),
            (1, (1,)),
        ]:
            with self.subTest(total_cents=total_cents, weights=weights):
                with self.assertRaises(ValueError):
                    allocate_cents(total_cents, weights)
