import unittest
from config import server_port

class ConfigTests(unittest.TestCase):
    def test_app_port_is_used(self):
        self.assertEqual(server_port({'APP_PORT': '9001'}), 9001)

    def test_absent_app_port_defaults_and_port_is_ignored(self):
        self.assertEqual(server_port({}), 8080)
        self.assertEqual(server_port({'PORT': '9001'}), 8080)
        self.assertEqual(server_port({'APP_PORT': '9001', 'PORT': '8000'}), 9001)

    def test_ascii_decimal_boundaries_are_accepted(self):
        self.assertEqual(server_port({'APP_PORT': '1'}), 1)
        self.assertEqual(server_port({'APP_PORT': '65535'}), 65535)
        self.assertEqual(server_port({'APP_PORT': '0001'}), 1)

    def test_invalid_present_app_port_values_raise_value_error(self):
        invalid_values = [None, '', ' 1', '1 ', '+1', '-1', '0', '65536', 'abc', '１２', 1]
        for value in invalid_values:
            with self.subTest(value=value):
                with self.assertRaises(ValueError):
                    server_port({'APP_PORT': value})
